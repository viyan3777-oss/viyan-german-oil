    try {
      localStorage.setItem('viyan_pro_branding', JSON.stringify(nextBranding));
      localStorage.setItem('viyan_pro_catalog', JSON.stringify(nextCatalog));
      localStorage.setItem('viyan_pro_oils', JSON.stringify(nextOils));
    } catch (err) {
      console.error('Save config error:', err);
    }

  const handleLogoUpload = (e) => {
    const file = e.target.files?.[0];
    if (!file) return;
    const reader = new FileReader();
    reader.onload = () => {
      const nextBranding = { ...branding, logoDataUrl: reader.result };
      setBranding(nextBranding);
      saveConfigToStore(nextBranding, catalogItems, oils);
      showNotification('Logo erfolgreich auf Kassenbon übernommen!');
    };
    reader.readAsDataURL(file);
  };

    e.preventDefault();
    if (!newItemDe.trim() || !newItemPrice) return;
    const item = {
      id: 'item-' + Date.now(),
      category: newItemCategory.trim() || 'Service',
      de: newItemDe.trim(),
      ar: newItemAr.trim() || newItemDe.trim(),
      costEur: Number(newItemCost) || 0,
      priceEur: Number(newItemPrice) || 0
    };
    const nextCatalog = [...catalogItems, item];
    setCatalogItems(nextCatalog);
    saveConfigToStore(branding, nextCatalog, oils);

    setNewItemDe('');
    setNewItemAr('');
    setNewItemCost('');
    setNewItemPrice('');
    showNotification(`"${item.de}" zum Kassen-Katalog hinzugefügt!`);
  };

  const handleDeleteCatalogItem = (id) => {
    const nextCatalog = catalogItems.filter((item) => item.id !== id);
    setCatalogItems(nextCatalog);
    saveConfigToStore(branding, nextCatalog, oils);
    showNotification('Artikel aus Katalog entfernt.', 'info');
  };

  const handleUpdateOilPrice = (oilId, field, value) => {
    const nextOils = oils.map((o) =>
      o.id === oilId ? { ...o, [field]: Number(value) || 0 } : o
    );
    setOils(nextOils);
    saveConfigToStore(branding, catalogItems, nextOils);
  };

  // Cart Quantity Helpers
  const updateItemQty = (itemId, delta) => {
    setCartQuantities((prev) => {
      const current = prev[itemId] || 0;
      const next = Math.max(0, current + delta);
      const copy = { ...prev };
      if (next === 0) {
        delete copy[itemId];
      } else {
        copy[itemId] = next;
      }
      return copy;
    });
  };

  const handleAddOneOff = () => {
    if (!oneOffName.trim() || !oneOffPrice) return;
    setOneOffItems((prev) => [
      ...prev,
      {
        id: 'custom-' + Date.now(),
        nameDe: oneOffName.trim(),
        nameAr: oneOffName.trim(),
        qty: 1,
        priceEur: Number(oneOffPrice) || 0,
        costEur: Number(oneOffCost) || 0
      }
    ]);
    setOneOffName('');
    setOneOffPrice('');
    setOneOffCost('');
    showNotification('Sonderposition zum Bon hinzugefügt.');
  };

  // Current Order Totals
  const selectedOil = selectedOilId === 'none' ? null : oils.find((o) => o.id === selectedOilId) || oils[0];

  const currentTotals = useMemo(() => {
    const oilPrice = selectedOil ? selectedOil.priceEur * Number(liters || 0) : 0;
    const oilCost = selectedOil ? selectedOil.costEur * Number(liters || 0) : 0;

    let itemsPrice = 0;
    let itemsCost = 0;
    const lineItems = [];

    Object.entries(cartQuantities).forEach(([itemId, qty]) => {
      const found = catalogItems.find((c) => c.id === itemId);
      if (found && qty > 0) {
        itemsPrice += found.priceEur * qty;
        itemsCost += found.costEur * qty;
        lineItems.push({
          nameDe: found.de,
          nameAr: found.ar,
          qty,
          priceEur: Number((found.priceEur * qty).toFixed(2)),
          costEur: Number((found.costEur * qty).toFixed(2))
        });

      itemsPrice += custom.priceEur * custom.qty;
      itemsCost += custom.costEur * custom.qty;
      lineItems.push(custom);

    const totalEur = Math.max(0, Number((oilPrice + itemsPrice - discount).toFixed(2)));

    const profitEur = Number((totalEur - costEur).toFixed(2));
    const totalIqd = Math.round(totalEur * (branding.iqdRate || 1430));

      (p) =>
        p.ar.includes(noteRaw.trim()) ||
        p.de.toLowerCase().includes(noteRaw.trim().toLowerCase())
    );
    if (match) {
      setNoteTranslated(`${match.de} | ${match.ar}`);
    } else {
      const hasArabic = /[\u0600-\u06FF]/.test(noteRaw);
      setNoteTranslated(hasArabic ? `[DE]: ${noteRaw}` : `[AR]: ${noteRaw}`);
    }

        lang === 'de' ? 'Bitte Autokennzeichen eingeben!' : 'يرجى إدخال رقم السيارة!',

      id: 'BON-' + Date.now().toString().slice(-6),
      plate: plate.trim().toUpperCase(),
      mileage: Number(mileage) || 0,
      phone: phone.trim(),
      pit: selectedPit,
      oilId: selectedOil ? selectedOil.id : 'none',
      oilName: selectedOil ? selectedOil.name : 'Nur Teile / Reifen / Service',
      liters: usedLiters,
      lineItems: currentTotals.lineItems,
      discountEur: currentTotals.discount,
      paymentMethod,
      totalEur: currentTotals.totalEur,
      costEur: currentTotals.costEur,
      profitEur: currentTotals.profitEur,
      noteRaw,
      noteTranslated: noteTranslated || noteRaw,
      createdAt: new Date().toISOString(),
      closedInZBon: false
    };

    const updatedDrums = { ...drumLevels };
    if (selectedOil && updatedDrums[selectedOil.id] !== undefined) {
      updatedDrums[selectedOil.id] = Math.max(
        0,
        Number((updatedDrums[selectedOil.id] - usedLiters).toFixed(1))
      );
    }

    const nextOrders = [newOrder, ...orders];
    setOrders(nextOrders);
    setDrumLevels(updatedDrums);
    setLastReceipt(newOrder);

    try {
      localStorage.setItem('viyan_pro_orders', JSON.stringify(nextOrders));
      localStorage.setItem('viyan_pro_drums', JSON.stringify(updatedDrums));
    } catch (err) {
      console.error(err);
    }

    showNotification(`Bon ${newOrder.id} verbucht (${newOrder.totalEur.toFixed(2)} €)!`);

    // Reset POS Form
    setPlate('');
    setMileage('');
    setPhone('+964 ');
    setCartQuantities({});
    setOneOffItems([]);
    setDiscountEur(0);
    setNoteRaw('');
    setNoteTranslated('');
  };

    const updatedDrums = { ...drumLevels, [oilId]: 208 };

    localStorage.setItem('viyan_pro_byproducts', JSON.stringify(nextBy));

  // Shift Metrics
  const openOrders = useMemo(() => orders.filter((o) => !o.closedInZBon), [orders]);
  const openByproducts = useMemo(() => byproducts.filter((b) => !b.closedInZBon), [byproducts]);

  const shiftStats = useMemo(() => {
    const oilRevenue = openOrders.reduce((sum, o) => sum + (o.totalEur || 0), 0);
    const oilProfit = openOrders.reduce((sum, o) => sum + (o.profitEur || 0), 0);
    const byproductRevenue = openByproducts.reduce((sum, b) => sum + (b.amountEur || 0), 0);

    const cashTotal =
      openOrders
        .filter((o) => o.paymentMethod !== 'card')
        .reduce((sum, o) => sum + (o.totalEur || 0), 0) + byproductRevenue;

    const cardTotal = openOrders
      .filter((o) => o.paymentMethod === 'card')
      .reduce((sum, o) => sum + (o.totalEur || 0), 0);

    const litersByOil = { '5w30': 0, '5w40': 0, '10w40': 0 };
    openOrders.forEach((o) => {
      if (litersByOil[o.oilId] !== undefined) {
        litersByOil[o.oilId] += Number(o.liters || 0);
      }
    });

    const totalRevenueEur = Number((oilRevenue + byproductRevenue).toFixed(2));
    return {
      totalRevenueEur,
      totalRevenueIqd: Math.round(totalRevenueEur * (branding.iqdRate || 1430)),
      totalProfitEur: Number((oilProfit + byproductRevenue).toFixed(2)),
      cashTotal: Number(cashTotal.toFixed(2)),
      cardTotal: Number(cardTotal.toFixed(2)),
      carCount: openOrders.length,
      byproductRevenue,
      litersByOil
    };
  }, [openOrders, openByproducts, branding.iqdRate]);

  // Create Z-Bon
  const handleCreateZBon = () => {
    if (openOrders.length === 0 && openByproducts.length === 0) {
      showNotification('Keine offenen Buchungen für einen Z-Bon vorhanden.', 'error');
      return;
    }

    const zNumber =
      'Z-' + new Date().toISOString().slice(0, 10).replace(/-/g, '') + '-' + (zBons.length + 1);

    const localZ = {
      id: zNumber,
      zNumber,
      createdAt: new Date().toISOString(),
      carCount: shiftStats.carCount,
      totalRevenueEur: shiftStats.totalRevenueEur,
      totalRevenueIqd: shiftStats.totalRevenueIqd,
      totalProfitEur: shiftStats.totalProfitEur,
      cashTotal: shiftStats.cashTotal,
      cardTotal: shiftStats.cardTotal,
      byproductRevenue: shiftStats.byproductRevenue,
      litersByOil: shiftStats.litersByOil
    };

    const nextZBons = [localZ, ...zBons];
    const nextOrders = orders.map((o) => ({ ...o, closedInZBon: true }));
    const nextBy = byproducts.map((b) => ({ ...b, closedInZBon: true }));

    setZBons(nextZBons);
    setOrders(nextOrders);
    setByproducts(nextBy);

    localStorage.setItem('viyan_pro_zbons', JSON.stringify(nextZBons));
    localStorage.setItem('viyan_pro_orders', JSON.stringify(nextOrders));
    localStorage.setItem('viyan_pro_byproducts', JSON.stringify(nextBy));
    showNotification(`Z-Bon ${zNumber} erfolgreich abgeschlossen!`);
  };

  const confirmDeleteOrder = () => {
    if (!isChefUnlocked || !deleteOrderId) return;
    const nextOrders = orders.filter((o) => o.id !== deleteOrderId);
    setOrders(nextOrders);
    localStorage.setItem('viyan_pro_orders', JSON.stringify(nextOrders));
    setDeleteOrderId(null);
    showNotification('Buchung vom Chef storniert.', 'info');
  };

  const buildWhatsappUrl = (order) => {
    const cleanPhone = (order.phone || '').replace(/[^0-9]/g, '');
    const nextKm = (Number(order.mileage) || 0) + 7000;
    const msg =
      `مرحباً من ${branding.companyName} 🇩🇪🇮🇶%0A` +
      `لقد حان موعد الصيانة وتبديل الزيت لسيارتك رقم (${order.plate}).%0A` +
      `آخر قراءة للعداد كانت: ${order.mileage} كم (الموعد القادم: ${nextKm} كم).%0A` +
      `العنوان: ${branding.address}%0A` +
      `نحن بانتظاركم!%0A%0A` +
      `Hallo von ${branding.companyName}! Zeit für Ihren nächsten Ölwechsel für Kennzeichen ${order.plate} (letzter KM: ${order.mileage} km).`;
    return `https://wa.me/${cleanPhone}?text=${msg}`;
  };

  const categories = useMemo(() => {
    const set = new Set(catalogItems.map((i) => i.category));
    return ['ALL', ...Array.from(set)];
  }, [catalogItems]);

  const filteredCatalog = useMemo(() => {
    if (selectedCategoryFilter === 'ALL') return catalogItems;
    return catalogItems.filter((i) => i.category === selectedCategoryFilter);
  }, [catalogItems, selectedCategoryFilter]);

  const filteredCustomers = useMemo(() => {
    if (!customerSearch.trim()) return orders;
    const q = customerSearch.toLowerCase();
    return orders.filter(
      (o) =>
        (o.plate || '').toLowerCase().includes(q) ||
        (o.phone || '').toLowerCase().includes(q)
    );
  }, [orders, customerSearch]);

  return (
    <div dir={isRtl ? 'rtl' : 'ltr'} className="min-h-screen bg-slate-950 text-slate-100 pb-16">
      {/* Print Styles */}
      <style>{`
        @media print {
          body * { visibility: hidden; }
          #printable-receipt, #printable-receipt * { visibility: visible; }
          #printable-receipt {
            position: fixed;
            left: 0;
            top: 0;
            width: 80mm;
            max-width: 100%;
            background: white !important;
            color: black !important;
            box-shadow: none !important;
            border: none !important;
          }
          .no-print { display: none !important; }
        }
      `}</style>

      {/* Toast Notification */}
      {toast && (
        <div
          className={`fixed bottom-5 right-5 z-50 px-5 py-3.5 rounded-2xl shadow-2xl flex items-center gap-2.5 text-sm font-bold border ${
            toast.type === 'error'
              ? 'bg-red-950 border-red-500 text-red-200'
              : 'bg-slate-900 border-amber-500 text-amber-300'
          }`}
        >
          {toast.type === 'error' ? (
            <AlertTriangle className="w-4 h-4 text-red-400 shrink-0" />
          ) : (
            <CheckCircle2 className="w-4 h-4 text-emerald-400 shrink-0" />
          )}
          <span>{toast.message}</span>
        </div>
      )}

      {/* Delete Confirmation Modal */}
      {deleteOrderId && (
        <div className="fixed inset-0 z-50 bg-black/75 flex items-center justify-center p-4">
          <div className="bg-slate-900 border border-slate-700 rounded-2xl p-6 max-w-sm w-full space-y-4">
            <div className="flex items-center justify-between">
              <h3 className="font-bold text-white flex items-center gap-2">
                <AlertTriangle className="w-5 h-5 text-red-400" /> Buchung stornieren?
              </h3>
              <button onClick={() => setDeleteOrderId(null)} className="text-slate-400 hover:text-white">
                <X className="w-5 h-5" />
              </button>
            </div>
            <p className="text-xs text-slate-400">
              Möchtest du den Bon <strong className="text-white">{deleteOrderId}</strong> wirklich endgültig löschen?
            </p>
            <div className="flex justify-end gap-2">
              <button
                onClick={() => setDeleteOrderId(null)}
                className="px-4 py-2 rounded-xl bg-slate-800 text-slate-300 text-xs font-bold"
              >
                Abbrechen
              </button>
              <button
                onClick={confirmDeleteOrder}
                className="px-4 py-2 rounded-xl bg-red-600 hover:bg-red-500 text-white text-xs font-bold"
              >
                Ja, löschen
              </button>
            </div>
          </div>
        </div>
      )}

      {/* Top Header */}
      <header className="bg-slate-900 border-b border-slate-800 sticky top-0 z-30">
        <div className="max-w-7xl mx-auto px-4 py-3 flex flex-wrap items-center justify-between gap-3">
          <div className="flex items-center gap-3">
            {branding.logoDataUrl ? (
              <img
                src={branding.logoDataUrl}
                alt="Logo"
                className="w-10 h-10 rounded-xl object-contain bg-white p-1"
              />
            ) : (
              <div className="w-10 h-10 rounded-xl bg-amber-500 flex items-center justify-center text-slate-950 font-black">
                <Droplet className="w-6 h-6" />
              </div>
            )}
            <div>
              <div className="flex items-center gap-2">
                <h1 className="font-bold text-lg tracking-tight text-white">{branding.companyName}</h1>
                <span className="text-[10px] font-bold px-2 py-0.5 rounded-full bg-amber-500/20 text-amber-400 border border-amber-500/30">
                  PRO v2.0
                </span>
              </div>
              <p className="text-xs text-slate-400">{branding.subtitle}</p>
            </div>
          </div>

          <div className="flex items-center gap-2">
            {isChefUnlocked && (
              <span className="text-xs bg-emerald-500/20 border border-emerald-500/40 text-emerald-300 px-3 py-1.5 rounded-xl font-bold flex items-center gap-1.5">
                <ShieldCheck className="w-3.5 h-3.5" /> Chef-Modus aktiv
              </span>
            )}
            <button
              onClick={() => setLang(lang === 'de' ? 'ar' : 'de')}
              className="flex items-center gap-2 bg-slate-800 hover:bg-slate-700 border border-slate-700 px-3.5 py-2 rounded-xl text-sm font-semibold text-amber-400 transition"
            >
              <Languages className="w-4 h-4" />
              {lang === 'de' ? 'العربية (AR)' : 'Deutsch (DE)'}
            </button>
          </div>
        </div>

        {/* Main Navigation */}
        <div className="max-w-7xl mx-auto px-4 flex gap-2 overflow-x-auto pb-2.5">
          {[
            { id: 'pos', label: t.posTab, icon: Receipt },
            { id: 'customers', label: t.customersTab, icon: MessageCircle },
            { id: 'zbon', label: t.zbonTab, icon: FileText },
            { id: 'chef', label: t.chefTab, icon: ShieldCheck }
          ].map((tab) => {
            const Icon = tab.icon;
            const active = activeTab === tab.id;
            return (
              <button
                key={tab.id}
                onClick={() => setActiveTab(tab.id)}
                className={`flex items-center gap-2 px-4 py-2.5 rounded-xl text-sm font-semibold whitespace-nowrap transition ${
                  active
                    ? 'bg-amber-500 text-slate-950 shadow-lg shadow-amber-500/20'
                    : 'bg-slate-800/70 text-slate-300 hover:bg-slate-800'
                }`}
              >
                <Icon className="w-4 h-4" />
                {tab.label}
              </button>
            );
          })}
        </div>
      </header>

      <main className="max-w-7xl mx-auto px-4 pt-6">
        {/* TAB 1: POS / KASSE */}
        {activeTab === 'pos' && (
          <div className="grid grid-cols-1 lg:grid-cols-12 gap-6">
            <form onSubmit={handleCheckout} className="lg:col-span-8 space-y-6">
              {/* Vehicle Data & Pit */}
              <div className="bg-slate-900 border border-slate-800 rounded-2xl p-5 space-y-4">
                <div className="flex flex-wrap items-center justify-between gap-2">
                  <span className="text-sm font-bold text-amber-400 flex items-center gap-2">
                    <Car className="w-4 h-4" /> {t.vehicleHeader}
                  </span>
                  <div className="flex gap-2">
                    {[1, 2, 3].map((pitNum) => (
                      <button
                        type="button"
                        key={pitNum}
                        onClick={() => setSelectedPit(pitNum)}
                        className={`px-4 py-2 rounded-xl text-sm font-bold border transition ${
                          selectedPit === pitNum
                            ? 'bg-amber-500 text-slate-950 border-amber-400'
                            : 'bg-slate-950 text-slate-300 border-slate-700 hover:border-slate-500'
                        }`}
                      >
                        {t.pit} {pitNum}
                      </button>
                    ))}
                  </div>
                </div>

                <div className="grid grid-cols-1 sm:grid-cols-3 gap-4">
                  <div>
                    <label className="block text-xs text-slate-400 mb-1">{t.plateLabel}</label>
                    <input
                      type="text"
                      required
                      placeholder={t.platePlaceholder}
                      value={plate}
                      onChange={(e) => setPlate(e.target.value)}
                      className="w-full bg-slate-950 border border-slate-700 rounded-xl px-3.5 py-2.5 text-white font-bold focus:border-amber-500 focus:outline-none"
                    />
                  </div>
                  <div>
                    <label className="block text-xs text-slate-400 mb-1 flex items-center gap-1">
                      <Gauge className="w-3.5 h-3.5 text-amber-400" /> {t.kmLabel}
                    </label>
                    <input
                      type="number"
                      required
                      placeholder="z.B. 85000"
                      value={mileage}
                      onChange={(e) => setMileage(e.target.value)}
                      className="w-full bg-slate-950 border border-slate-700 rounded-xl px-3.5 py-2.5 text-white font-bold focus:border-amber-500 focus:outline-none"
                    />
                  </div>
                  <div>
                    <label className="block text-xs text-slate-400 mb-1 flex items-center gap-1">
                      <Phone className="w-3.5 h-3.5 text-emerald-400" /> {t.phoneLabel}
                    </label>
                    <input
                      type="tel"
                      required
                      placeholder="+964 750 ..."
                      value={phone}
                      onChange={(e) => setPhone(e.target.value)}
                      className="w-full bg-slate-950 border border-slate-700 rounded-xl px-3.5 py-2.5 text-white font-bold focus:border-amber-500 focus:outline-none"
                    />
                  </div>
                </div>
              </div>

              {/* Oil Selection (Including "No Oil / Parts Only" option!) */}
              <div className="bg-slate-900 border border-slate-800 rounded-2xl p-5 space-y-4">
                <div className="flex flex-wrap items-center justify-between gap-2">
                  <h2 className="text-sm font-bold text-amber-400">{t.selectOil}</h2>
                  <button
                    type="button"
                    onClick={() => setSelectedOilId('none')}
                    className={`text-xs px-3.5 py-2 rounded-xl font-bold border transition ${
                      selectedOilId === 'none'
                        ? 'bg-sky-500 text-slate-950 border-sky-400'
                        : 'bg-slate-950 text-slate-300 border-slate-700 hover:border-slate-500'
                    }`}
                  >
                    {t.noOilNeeded}
                  </button>
                </div>

                <div className="grid grid-cols-1 sm:grid-cols-3 gap-3">
                  {oils.map((oil) => {
                    const selected = oil.id === selectedOilId;
                    return (
                      <button
                        type="button"
                        key={oil.id}
                        onClick={() => setSelectedOilId(oil.id)}
                        className={`p-4 rounded-xl border text-start transition ${
                          selected
                            ? 'bg-amber-500/15 border-amber-500 ring-1 ring-amber-500'
                            : 'bg-slate-950 border-slate-800 hover:border-slate-700'
                        }`}
                      >
                        <div className="font-bold text-white text-sm">{oil.name}</div>
                        <div className="text-xs text-slate-400 mt-1">{oil.spec}</div>
                        <div className="mt-3 flex items-baseline justify-between">
                          <span className="text-amber-400 font-extrabold text-base">
                            {oil.priceEur.toFixed(2)} € / L
                          </span>
                          <span className="text-[11px] text-slate-400">
                            {Math.round(oil.priceEur * (branding.iqdRate || 1430)).toLocaleString()} IQD
                          </span>
                        </div>
                      </button>
                    );
                  })}
                </div>

                {selectedOilId !== 'none' && (
                  <div className="pt-2">
                    <label className="block text-xs text-slate-400 mb-2">{t.litersLabel}</label>
                    <div className="flex flex-wrap items-center gap-2">
                      {[4, 4.5, 5, 5.5, 6, 7, 8].map((preset) => (
                        <button
                          type="button"
                          key={preset}
                          onClick={() => setLiters(preset)}
                          className={`px-3.5 py-2 rounded-xl text-sm font-bold border ${
                            Number(liters) === preset
                              ? 'bg-amber-500 text-slate-950 border-amber-400'
                              : 'bg-slate-950 text-slate-300 border-slate-700'
                          }`}
                        >
                          {preset} L
                        </button>
                      ))}
                      <input
                        type="number"
                        step="0.1"
                        min="0.5"
                        max="50"
                        value={liters}
                        onChange={(e) => setLiters(e.target.value)}
                        className="w-28 bg-slate-950 border border-amber-500/60 rounded-xl px-3 py-2 text-center font-extrabold text-amber-400"
                      />
                    </div>
                  </div>
                )}
              </div>

              {/* Dynamic Catalog: Filters, Tires, Parts & Services with Quantity Stepper */}
              <div className="bg-slate-900 border border-slate-800 rounded-2xl p-5 space-y-4">
                <div className="flex flex-wrap items-center justify-between gap-2">
                  <h2 className="text-sm font-bold text-amber-400 flex items-center gap-2">
                    <Wrench className="w-4 h-4" /> {t.extrasLabel}
                  </h2>
                  <div className="flex flex-wrap gap-1.5">
                    {categories.map((cat) => (
                      <button
                        type="button"
                        key={cat}
                        onClick={() => setSelectedCategoryFilter(cat)}
                        className={`text-xs px-3 py-1.5 rounded-lg font-bold transition ${
                          selectedCategoryFilter === cat
                            ? 'bg-amber-500 text-slate-950'
                            : 'bg-slate-950 text-slate-400 border border-slate-800'
                        }`}
                      >
                        {cat === 'ALL' ? (lang === 'de' ? 'Alle' : 'الكل') : cat}
                      </button>
                    ))}
                  </div>
                </div>

                <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
                  {filteredCatalog.map((item) => {
                    const qty = cartQuantities[item.id] || 0;
                    return (
                      <div
                        key={item.id}
                        className={`flex items-center justify-between p-3.5 rounded-xl border transition ${
                          qty > 0
                            ? 'bg-emerald-500/15 border-emerald-500 text-white'
                            : 'bg-slate-950 border-slate-800 text-slate-300'
                        }`}
                      >
                        <div className="pr-2">
                          <span className="text-[10px] uppercase tracking-wider px-2 py-0.5 rounded bg-slate-800 text-amber-400 font-bold">
                            {item.category}
                          </span>
                          <div className="text-sm font-bold mt-1">
                            {lang === 'de' ? item.de : item.ar}
                          </div>
                          <div className="text-xs font-bold text-emerald-400">
                            {item.priceEur.toFixed(2)} € / Stk.
                          </div>
                        </div>

                        <div className="flex items-center gap-2">
                          <button
                            type="button"
                            onClick={() => updateItemQty(item.id, -1)}
                            className="w-9 h-9 rounded-xl bg-slate-800 hover:bg-slate-700 font-black text-white"
                          >
                            -
                          </button>
                          <span className="w-6 text-center font-black text-sm">{qty}</span>
                          <button
                            type="button"
                            onClick={() => updateItemQty(item.id, 1)}
                            className="w-9 h-9 rounded-xl bg-emerald-600 hover:bg-emerald-500 font-black text-white"
                          >
                            +
                          </button>
                        </div>
                      </div>
                    );
                  })}
                </div>

                {/* Quick Ad-Hoc Custom Item */}
                <div className="pt-3 border-t border-slate-800">
                  <div className="text-xs font-bold text-slate-400 mb-2">{t.customOneOffTitle}</div>
                  <div className="flex flex-wrap gap-2">
                    <input
                      type="text"
                      placeholder="Bezeichnung (z.B. Scheibenwischer / Sonderarbeit)"
                      value={oneOffName}
                      onChange={(e) => setOneOffName(e.target.value)}
                      className="flex-1 min-w-[200px] bg-slate-950 border border-slate-700 rounded-xl px-3 py-2 text-xs text-white"
                    />
                    <input
                      type="number"
                      step="0.5"
                      placeholder="EK (€)"
                      value={oneOffCost}
                      onChange={(e) => setOneOffCost(e.target.value)}
                      className="w-20 bg-slate-950 border border-slate-700 rounded-xl px-2.5 py-2 text-xs text-slate-300"
                    />
                    <input
                      type="number"
                      step="0.5"
                      placeholder="VK (€)"
                      value={oneOffPrice}
                      onChange={(e) => setOneOffPrice(e.target.value)}
                      className="w-24 bg-slate-950 border border-slate-700 rounded-xl px-2.5 py-2 text-xs font-bold text-emerald-400"
                    />
                    <button
                      type="button"
                      onClick={handleAddOneOff}
                      className="bg-slate-800 hover:bg-slate-700 text-amber-400 font-bold text-xs px-3.5 py-2 rounded-xl flex items-center gap-1"
                    >
                      <Plus className="w-4 h-4" /> Hinzufügen
                    </button>
                  </div>
                  {oneOffItems.length > 0 && (
                    <div className="mt-2 flex flex-wrap gap-2">
                      {oneOffItems.map((c) => (
                        <span
                          key={c.id}
                          className="text-xs bg-emerald-950/60 border border-emerald-500/40 text-emerald-300 px-3 py-1 rounded-lg flex items-center gap-2"
                        >
                          {c.nameDe}: {c.priceEur.toFixed(2)} €
                          <button
                            type="button"
                            onClick={() => setOneOffItems(oneOffItems.filter((x) => x.id !== c.id))}
                            className="text-red-400 font-bold"
                          >
                            ×
                          </button>
                        </span>
                      ))}
                    </div>
                  )}
                </div>
              </div>

              {/* Payment Method, Discount & Bilingual Notes */}
              <div className="bg-slate-900 border border-slate-800 rounded-2xl p-5 space-y-4">
                <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
                  <div>
                    <label className="block text-xs font-bold text-slate-400 mb-1.5">
                      {t.paymentMethodLabel}
                    </label>
                    <div className="grid grid-cols-2 gap-2">
                      <button
                        type="button"
                        onClick={() => setPaymentMethod('cash')}
                        className={`py-2.5 px-3 rounded-xl text-xs font-bold border flex items-center justify-center gap-1.5 ${
                          paymentMethod === 'cash'
                            ? 'bg-emerald-500/20 border-emerald-500 text-emerald-300'
                            : 'bg-slate-950 border-slate-800 text-slate-400'
                        }`}
                      >
                        <Wallet className="w-4 h-4" /> {t.cash}
                      </button>
                      <button
                        type="button"
                        onClick={() => setPaymentMethod('card')}
                        className={`py-2.5 px-3 rounded-xl text-xs font-bold border flex items-center justify-center gap-1.5 ${
                          paymentMethod === 'card'
                            ? 'bg-sky-500/20 border-sky-500 text-sky-300'
                            : 'bg-slate-950 border-slate-800 text-slate-400'
                        }`}
                      >
                        <CreditCard className="w-4 h-4" /> {t.card}
                      </button>
                    </div>
                  </div>

                  <div>
                    <label className="block text-xs font-bold text-slate-400 mb-1.5 flex items-center gap-1">
                      <Percent className="w-3.5 h-3.5 text-amber-400" /> {t.discountLabel}
                    </label>
                    <input
                      type="number"
                      min="0"
                      step="0.5"
                      value={discountEur}
                      onChange={(e) => setDiscountEur(e.target.value)}
                      className="w-full bg-slate-950 border border-slate-700 rounded-xl px-3.5 py-2 text-sm font-bold text-amber-400"
                    />
                  </div>
                </div>

                {/* Notes */}
                <div className="space-y-2 pt-2 border-t border-slate-800">
                  <div className="flex items-center justify-between">
                    <label className="text-xs font-bold text-amber-400">{t.notesLabel}</label>
                    <button

                            onChange={(e) => setNewItemAr(e.target.value)}
                            className="w-full bg-slate-950 border border-slate-700 rounded-xl px-3 py-2.5 text-xs text-white"
                          />
                        </div>
                        <div>
                          <label className="block text-[11px] text-slate-400 mb-1">Dein EK (€)</label>
                          <input
                            type="number"
                            step="0.1"
                            required
                            placeholder="45.00"
                            value={newItemCost}
                            onChange={(e) => setNewItemCost(e.target.value)}
                            className="w-full bg-slate-950 border border-slate-700 rounded-xl px-3 py-2.5 text-xs text-slate-300"
                          />
                        </div>
                        <div>
                          <label className="block text-[11px] text-slate-400 mb-1">Kunden-VK (€)</label>
                          <input
                            type="number"
                            step="0.1"
                            required
                            placeholder="68.00"
                            value={newItemPrice}
                            onChange={(e) => setNewItemPrice(e.target.value)}
                            className="w-full bg-slate-950 border border-slate-700 rounded-xl px-3 py-2.5 text-xs font-bold text-emerald-400"
                          />
                        </div>
                        <div className="sm:col-span-6 flex justify-end">
                          <button
                            type="submit"
                            className="bg-amber-500 hover:bg-amber-400 text-slate-950 font-bold px-5 py-2.5 rounded-xl text-xs flex items-center gap-1.5"
                          >
                            <Plus className="w-4 h-4" /> Jetzt in Kasse speichern
                          </button>
                        </div>
                      </form>
                    </div>

                    {/* Oil Prices Editor */}
                    <div className="bg-slate-900 border border-slate-800 rounded-2xl p-5 space-y-3">
                      <h3 className="text-sm font-bold text-white">Öl-Literpreise (EK & VK) anpassen</h3>
                      <div className="grid grid-cols-1 sm:grid-cols-3 gap-3">
                        {oils.map((o) => (
                          <div key={o.id} className="bg-slate-950 border border-slate-800 rounded-xl p-3.5 space-y-2">
                            <div className="font-bold text-xs text-amber-400">{o.name}</div>
                            <div className="flex gap-2">
                              <div className="flex-1">
                                <label className="text-[10px] text-slate-400">EK (€/L)</label>
                                <input
                                  type="number"
                                  step="0.1"
                                  value={o.costEur}
                                  onChange={(e) => handleUpdateOilPrice(o.id, 'costEur', e.target.value)}
                                  className="w-full bg-slate-900 border border-slate-700 rounded-lg px-2 py-1 text-xs text-white"
                                />
                              </div>
                              <div className="flex-1">
                                <label className="text-[10px] text-slate-400">VK (€/L)</label>
                                <input
                                  type="number"
                                  step="0.1"
                                  value={o.priceEur}
                                  onChange={(e) => handleUpdateOilPrice(o.id, 'priceEur', e.target.value)}
                                  className="w-full bg-slate-900 border border-slate-700 rounded-lg px-2 py-1 text-xs font-bold text-emerald-400"
                                />
                              </div>
                            </div>
                          </div>
                        ))}
                      </div>
                    </div>

                    {/* Active Catalog List */}
                    <div className="bg-slate-900 border border-slate-800 rounded-2xl p-5 space-y-3">
                      <h3 className="text-sm font-bold text-white">
                        Aktive Teile, Reifen & Services ({catalogItems.length})
                      </h3>
                      <div className="space-y-2">
                        {catalogItems.map((item) => (
                          <div
                            key={item.id}
                            className="bg-slate-950 border border-slate-800 rounded-xl p-3 flex items-center justify-between text-xs"
                          >
                            <div>
                              <span className="px-2 py-0.5 rounded bg-slate-800 text-amber-400 font-bold mr-2">
                                {item.category}
                              </span>
                              <strong className="text-white">{item.de}</strong> /{' '}
                              <span className="text-slate-400">{item.ar}</span>
                            </div>
                            <div className="flex items-center gap-4">
                              <span>
                                EK: <strong>{item.costEur.toFixed(2)} €</strong> • VK:{' '}
                                <strong className="text-emerald-400">{item.priceEur.toFixed(2)} €</strong>
                              </span>
                              <button
                                onClick={() => handleDeleteCatalogItem(item.id)}
                                className="text-slate-500 hover:text-red-400"
                              >
                                <Trash2 className="w-4 h-4" />
                              </button>
                            </div>
                          </div>
                        ))}
                      </div>
                    </div>
                  </div>
                )}

                {/* SUB-TAB C: BRANDING, LOGO & RECEIPT ADDRESS */}
                {chefSubTab === 'branding' && (
                  <div className="bg-slate-900 border border-slate-800 rounded-2xl p-5 space-y-5 max-w-2xl">
                    <h3 className="text-base font-bold text-amber-400 flex items-center gap-2">
                      <ImageIcon className="w-5 h-5" /> Firmen-Logo & Adresse für Kassenbons einstellen
                    </h3>

                    <div className="space-y-4">
                      <div>
                        <label className="block text-xs font-bold text-slate-300 mb-1.5">
                          Eigenes Logo hochladen (erscheint oben auf jedem gedruckten Bon)
                        </label>
                        <div className="flex items-center gap-4">
                          {branding.logoDataUrl && (
                            <img
                              src={branding.logoDataUrl}
                              alt="Vorschau"
                              className="h-16 w-16 object-contain bg-white rounded-xl p-1 border border-slate-700"
                            />
                          )}
                          <input
                            type="file"
                            accept="image/*"
                            onChange={handleLogoUpload}
                            className="text-xs text-slate-300 file:mr-3 file:py-2 file:px-4 file:rounded-xl file:border-0 file:text-xs file:font-bold file:bg-amber-500 file:text-slate-950 hover:file:bg-amber-400"
                          />
                        </div>
                      </div>

                      <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
                        <div>
                          <label className="block text-xs text-slate-400 mb-1">Firmenname auf dem Bon</label>
                          <input
                            type="text"
                            value={branding.companyName}
                            onChange={(e) => setBranding({ ...branding, companyName: e.target.value })}
                            className="w-full bg-slate-950 border border-slate-700 rounded-xl px-3.5 py-2 text-sm text-white"
                          />
                        </div>
                        <div>
                          <label className="block text-xs text-slate-400 mb-1">Werkstatt-Telefon auf dem Bon</label>
                          <input
                            type="text"
                            value={branding.phone}
                            onChange={(e) => setBranding({ ...branding, phone: e.target.value })}
                            className="w-full bg-slate-950 border border-slate-700 rounded-xl px-3.5 py-2 text-sm text-white"
                          />
                        </div>
                      </div>

                      <div>
                        <label className="block text-xs text-slate-400 mb-1 flex items-center gap-1">
                          <MapPin className="w-3.5 h-3.5 text-amber-400" /> Werkstatt-Adresse in Erbil
                        </label>
                        <input
                          type="text"
                          value={branding.address}
                          onChange={(e) => setBranding({ ...branding, address: e.target.value })}
                          className="w-full bg-slate-950 border border-slate-700 rounded-xl px-3.5 py-2 text-sm text-white"
                        />
                      </div>

                      <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
                        <div>
                          <label className="block text-xs text-slate-400 mb-1">Fußzeile / Grußtext auf Bon</label>
                          <input
                            type="text"
                            value={branding.footerNote}
                            onChange={(e) => setBranding({ ...branding, footerNote: e.target.value })}
                            className="w-full bg-slate-950 border border-slate-700 rounded-xl px-3.5 py-2 text-sm text-white"
                          />
                        </div>
                        <div>
                          <label className="block text-xs text-slate-400 mb-1">Wechselkurs (1 € in IQD)</label>
                          <input
                            type="number"
                            value={branding.iqdRate}
                            onChange={(e) =>
                              setBranding({ ...branding, iqdRate: Number(e.target.value) || 1430 })
                            }
                            className="w-full bg-slate-950 border border-slate-700 rounded-xl px-3.5 py-2 text-sm font-bold text-amber-400"
                          />
                        </div>
                      </div>

                      <button
                        type="button"
                        onClick={() => {
                          saveConfigToStore(branding, catalogItems, oils);
                          showNotification('Logo, Adresse & Einstellungen gespeichert!');
                        }}
                        className="bg-emerald-600 hover:bg-emerald-500 text-white font-bold px-6 py-3 rounded-xl text-sm"
                      >
                        Änderungen speichern
                      </button>
                    </div>
                  </div>
                )}
              </div>
            )}
          </div>
        )}
      </main>
    </div>
  );
}
